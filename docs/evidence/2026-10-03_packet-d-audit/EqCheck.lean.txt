def labelEq (a b : StepLabel) : Bool := a.trace == b.trace && a.picks == b.picks && a.out == b.out

def resEq (a b : R) : Bool :=
  match a, b with
  | .ok (c, s, ch, l), .ok (c', s', ch', l') => Config.eqbF 100000 c c' && Store.eqb s s' && ch == ch' && labelEq l l'
  | .error e, .error e' => e == e'
  | _, _ => false

def errOf (a : R) : String := match a with | .error e => reprStr e | .ok _ => "<ok>"
